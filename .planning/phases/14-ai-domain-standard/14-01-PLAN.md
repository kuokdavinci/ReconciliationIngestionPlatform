<objective>
**AI Output Guardrails & Hallucination Detection**

Implement post-LLM validation that cross-references each generated insight's numeric claims against the `AnalysisInput` data. This closes the top two EVAL-REVIEW critical gaps: (1) hallucination detection, and (2) online guardrails.

**Purpose:** Before any LLM-generated insight reaches the API response, a guardrail verifies that numeric claims (mismatch_rate, affected_count, severity thresholds) are supported by input data. Unsupported insights are flagged, downgraded, or excluded.

**Output:** Enhanced `insights.py` orchestration with guardrail integration, guardrail verification functions in `services.py`, new model in `schemas.py`, and tests.
</objective>

    Add AFTER existing functions, BEFORE the `rule_based_pre_process` section:
    
    ```python
    import re as _re  # Move import re to module level (fixes IN-01)
    
    # ... keep existing _extract_json_from_response as-is ...

    # --- Guardrail verification helpers ---

    def _extract_percentage_from_text(text: str) -> Optional[float]:
        """Extract percentage values from insight text."""
        match = _re.search(r"(\d+\.?\d*)%", text)
        if match:
            return float(match.group(1))
        return None

    def verify_insight_claims(
        insight: AnalysisResult,
        analysis_input: AnalysisInput,
    ) -> GuardrailResult:
        """Cross-reference an insight's numeric claims against AnalysisInput data.
        
        Args:
            insight: AnalysisResult from LLM to verify.
            analysis_input: Original structured input with ground truth metrics.
        
        Returns:
            GuardrailResult with supported flag and any issues found.
        """
        issues: list[str] = []
        severity = insight.severity.lower()
        adjusted = severity
        metrics = analysis_input.summary_metrics
        total_tx = metrics.get("total_transactions", 0)
        mismatch_rate = metrics.get("mismatch_rate", 0.0)

        # 1. Verify affected_count doesn't exceed total_transactions
        if insight.affected_count > total_tx > 0:
            issues.append(f"affected_count ({insight.affected_count}) exceeds total_transactions ({total_tx})")

        # 2. Verify severity matches data thresholds (downgrade if too high)
        if severity == "critical":
            if mismatch_rate < 10.0 and insight.affected_count < 50:
                adjusted = "high"
                issues.append(f"severity downgraded from critical to high: mismatch_rate={mismatch_rate}% < 10% AND affected_count={insight.affected_count} < 50")
            elif mismatch_rate < 10.0:
                adjusted = "high"
                issues.append(f"severity downgraded from critical to high: mismatch_rate={mismatch_rate}% < 10%")
            elif insight.affected_count < 50:
                adjusted = "high"
                issues.append(f"severity downgraded from critical to high: affected_count={insight.affected_count} < 50")
        elif severity == "high":
            if mismatch_rate < 5.0 and insight.affected_count < 20:
                adjusted = "medium"
                issues.append(f"severity downgraded from high to medium: mismatch_rate={mismatch_rate}% < 5% AND affected_count={insight.affected_count} < 20")
            elif mismatch_rate < 5.0:
                if insight.affected_count >= 20:
                    pass  # Keep high - count justifies it
                else:
                    adjusted = "medium"
                    issues.append(f"severity downgraded from high to medium: mismatch_rate={mismatch_rate}% < 5%")
            elif insight.affected_count < 20:
                adjusted = "medium"
                issues.append(f"severity downgraded from high to medium: affected_count={insight.affected_count} < 20")
        elif severity == "medium":
            if insight.affected_count < 5 and mismatch_rate < 1.0:
                adjusted = "low"
                issues.append(f"severity downgraded from medium to low: affected_count={insight.affected_count} < 5 AND mismatch_rate={mismatch_rate}% < 1%")

        # 3. Check if title or description mentions a percentage that contradicts metrics
        text_to_check = f"{insight.title} {insight.description}"
        claimed_pct = _extract_percentage_from_text(text_to_check)
        if claimed_pct is not None and mismatch_rate > 0 and insight.type != "mismatch_rate":
            # For non-mismatch-rate insights, check if claimed % is wildly different
            pass  # Too much semantic analysis needed — skip for code-based guardrail

        # For mismatch_rate type insights, verify the claimed rate
        if insight.type == "mismatch_rate" and mismatch_rate > 0:
            claimed_pct = _extract_percentage_from_text(text_to_check)
            if claimed_pct is not None:
                ratio = claimed_pct / mismatch_rate if mismatch_rate > 0 else 1.0
                if ratio > 1.5 or ratio < 0.5:
                    issues.append(f"claimed mismatch rate ({claimed_pct}%) deviates from actual ({mismatch_rate}%) by >50%")
                    adjusted = "low"

        supported = len(issues) == 0
        return GuardrailResult(
            insight=insight,
            supported=supported,
            issues=issues,
            adjusted_severity=adjusted if not supported else severity,
        )

    def verify_insight_batch(
        insights: list[AnalysisResult],
        analysis_input: AnalysisInput,
    ) -> tuple[list[AnalysisResult], list[GuardrailResult]]:
        """Verify a batch of insights against analysis input.
        
        Args:
            insights: List of AnalysisResult objects from LLM.
            analysis_input: Original structured input data.
        
        Returns:
            Tuple of (passed_insights, failed_guardrail_results).
            Passed insights may have downgraded severity.
            Failed insights are returned as GuardrailResults for logging.
        """
        passed: list[AnalysisResult] = []
        failed: list[GuardrailResult] = []

        for insight in insights:
            result = verify_insight_claims(insight, analysis_input)
            if result.supported:
                # Apply adjusted severity if any
                if result.adjusted_severity and result.adjusted_severity != insight.severity:
                    insight.severity = result.adjusted_severity
                passed.append(insight)
            else:
                failed.append(result)
                # Still include insights with non-critical issues, but with adjusted severity
                if any("severity downgraded" in issue for issue in result.issues):
                    insight.severity = result.adjusted_severity
                    passed.append(insight)
                # Insights with critical issues (e.g., affected_count > total) are excluded

        return passed, failed
    ```
