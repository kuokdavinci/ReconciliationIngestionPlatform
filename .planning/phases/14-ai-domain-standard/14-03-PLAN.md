    Add a new test class at the end of the file:
    ```python
    @pytest.mark.e2e
    class TestReferenceDataset:
        """Run reference dataset scenarios against the analysis pipeline."""
        
        REFERENCE_DATASET = Path(__file__).parent / "data" / "eval_reference.jsonl"
        
        def _load_scenarios(self) -> list[dict]:
            """Load scenarios from reference dataset JSONL."""
            scenarios = []
            with open(self.REFERENCE_DATASET) as f:
                for line in f:
                    line = line.strip()
                    if line:
                        scenarios.append(json.loads(line))
            return scenarios
        
        def _build_results(self, transactions: list[dict]) -> list[SimpleNamespace]:
            """Build result objects from transaction data."""
            results = []
            for txn in transactions:
                r = SimpleNamespace()
                r.partner = "MOMO"
                r.date = "2024-07-07"
                r.partner_amount = Decimal(str(txn.get("partner_amount", "0"))) if txn.get("partner_amount") else None
                r.internal_amount = Decimal(str(txn.get("internal_amount", "0"))) if txn.get("internal_amount") else None
                status = txn.get("reconciliation_status", "MATCHED")
                r.reconciliation_status = SimpleNamespace(value=status)
                results.append(r)
            return results
        
        @pytest.mark.parametrize("scenario_idx", range(10))
        def test_scenario_metrics(self, scenario_idx: int):
            """Verify MetricsService produces expected metrics for each scenario."""
            scenarios = self._load_scenarios()
            assert scenario_idx < len(scenarios), f"Scenario {scenario_idx} not found in dataset"
            scenario = scenarios[scenario_idx]
            
            results = self._build_results(scenario["input"]["transactions"])
            summary = MetricsService.compute_summary(results, scenario["partner"], scenario["date"])
            
            expected = scenario["expected_metrics"]
            assert summary.total_transactions == expected["total_transactions"], (
                f"[{scenario['id']}] expected {expected['total_transactions']} transactions, got {summary.total_transactions}"
            )
            assert summary.matched == expected["matched"], (
                f"[{scenario['id']}] expected {expected['matched']} matched, got {summary.matched}"
            )
            assert abs(summary.mismatch_rate - expected["mismatch_rate"]) < 0.01, (
                f"[{scenario['id']}] expected mismatch_rate={expected['mismatch_rate']}, got {summary.mismatch_rate}"
            )
        
        def test_all_scenarios_loaded(self):
            """Verify all 10 scenarios are loadable."""
            scenarios = self._load_scenarios()
            assert len(scenarios) >= 8, f"Expected at least 8 scenarios, got {len(scenarios)}"
            
            for s in scenarios:
                assert "id" in s, f"Scenario missing 'id': {s}"
                assert "input" in s, f"Scenario {s['id']} missing 'input'"
                assert "transactions" in s["input"], f"Scenario {s['id']} missing 'input.transactions'"
                assert "expected_metrics" in s, f"Scenario {s['id']} missing 'expected_metrics'"
                assert "expected_finding_types" in s, f"Scenario {s['id']} missing 'expected_finding_types'"
    ```
