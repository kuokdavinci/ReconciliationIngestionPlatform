}

    function badge(status) {
      const cls = String(status || "").toLowerCase().replace(/\s+/g, "-");

      if (cls === "amount_mismatch" || cls === "status_mismatch" || cls === "multiple_mismatch") 
        return `<span class="badge unmatched">${status}</span>`;
      if (cls === "missing_internal" || cls === "missing_partner") 
        return `<span class="badge processing">${status}</span>`;
      if (cls === "matched_failed") return `<span class="badge failed">MATCHED FAILED</span>`;
      if (cls === "unmapped_skipped") return `<span class="badge warning">UNMAPPED SKIPPED</span>`;
      // Transaction processing statuses (NEW)

      if (cls === "refunded" || cls === "reversed") return `<span class="badge warning">${status.toUpperCase()}</span>`;
      return `<span class="badge">${status}</span>`;
