    ```python
    # Amount range filter
    if amount_min is not None or amount_max is not None:
        amount_query: dict = {}
        try:
            if amount_min is not None:
                # Decimal128 comparison: use string comparison since MongoDB stores Decimal128
                amount_query["$gte"] = Decimal128(amount_min)
            if amount_max is not None:
                amount_query["$lte"] = Decimal128(amount_max)
            query["partnerData.amount"] = amount_query
        except Exception as e:
            raise HTTPException(status_code=400, detail=f"Invalid amount value: {e}")

    # Date range filter (extends existing single-date filter)
    if date_from is not None or date_to is not None:
        date_query: dict = {}
        if date_from:
            try:
                dt_from = datetime.strptime(date_from, "%Y-%m-%d").replace(tzinfo=timezone.utc)
                date_query["$gte"] = dt_from
            except ValueError:
                raise HTTPException(status_code=400, detail=f"Invalid date_from format: '{date_from}'. Expected YYYY-MM-DD.")
        if date_to:
            try:
                dt_to = datetime.strptime(date_to, "%Y-%m-%d").replace(hour=23, minute=59, second=59, microsecond=999999, tzinfo=timezone.utc)
                date_query["$lte"] = dt_to
            except ValueError:
                raise HTTPException(status_code=400, detail=f"Invalid date_to format: '{date_to}'. Expected YYYY-MM-DD.")
        # If single-date filter also set, merge with $and
        if "reconciliationDate" in query:
            existing_date = query.pop("reconciliationDate")
            query["$and"] = [{"reconciliationDate": existing_date}, {"reconciliationDate": date_query}]
        else:
            query["reconciliationDate"] = date_query
    ```
